# Face verification.
#
# dlib is a C++ extension and publishes NO binary wheels — plain `dlib` is a
# source distribution that needs a cmake toolchain and several minutes of
# compilation. `dlib-bin` is the community prebuilt wheel of the same 19.24.6
# source, so the image builds in seconds without cmake.
#
# face-recognition declares `dlib` as a hard dependency, which would drag the
# source build back in, so it is installed with --no-deps in the Dockerfile
# alongside its own two real dependencies (Click, face_recognition_models).
#
# face_recognition_models carries the trained model files themselves (~100 MB:
# the ResNet encoder, the 68-point shape predictor, and the CNN detector). It
# is pinned because face-recognition looks those files up by name at import.
#
# numpy is pinned below 2.0 because dlib's compiled extension is built against
# the 1.x C ABI.
#
# setuptools is a runtime dependency, not a build one: face_recognition_models
# locates its model files with `pkg_resources.resource_filename`, and Python 3.12
# images no longer ship pkg_resources unless setuptools is installed.
#
# app/face.py imports these behind a guard: if they are unavailable the selfie
# endpoint returns 503 and the ID-number and document-image claim paths keep
# working, so the app runs either way.
numpy==1.26.4
setuptools==80.9.0
dlib-bin==19.24.6
Click>=6.0
face_recognition_models==0.3.0
#
# NOTE: `face-recognition` itself is deliberately NOT listed here. It declares a
# hard `dlib` dependency, so `pip install -r requirements.txt` would resolve and
# compile the source dlib — defeating dlib-bin above and adding a multi-minute
# cmake build. The Dockerfile installs it separately with --no-deps. Adding it
# back to this file will break the image build.
